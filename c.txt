---------------------------------------
--GLOBAL SETTINGS AND CONSTANTS
---------------------------------------
WIDTH=240
HEIGHT=136
BASE_SPEED=2

---------------------------------------
--OBJECTS / DATA
---------------------------------------
plr={
 x=112,
 y=80,
 sprite=1,
 flame=128,
 muzzle=0,
 lives=3,
 sugar=0
}
bul={
 x=112,
 y=-10,
 sprite=7
}
score=0
starfield={}

---------------------------------------
--MAIN LOOP
---------------------------------------
function TIC()
 --logic update
 update_starfield()
 plr_movement()
 plr_bullet()
 plr_flame()
 plr_muzzle()
  
 --graphics rendering
 cls(0)
 draw_starfield()
 spr(bul.sprite,bul.x,bul.y,0,1,0,0,1,2)
 spr(plr.sprite,plr.x,plr.y,0,1,0,0,2,2)
 spr(plr.flame,plr.x-1,plr.y+16,0,1,0,0,1,1)
 spr(plr.flame,plr.x+9,plr.y+16,0,1,0,0,1,1)
 if plr.muzzle>0 then
  circ(plr.x+8,plr.y-1,plr.muzzle,12)
 end
 draw_interface()
 print("SCORE: "..score,1,3,12,false,1,true)
 plr_health()
end

---------------------------------------
--STATE MACHINE
---------------------------------------

---------------------------------------
--FUNCTIONS
---------------------------------------
function plr_movement()
 local dx=0
 local dy=0
 
 if btn(3) or key(4)  then dx=dx+1 plr.sprite=5 end
 if btn(2) or key(1)  then dx=dx-1 plr.sprite=3 end
 if btn(1) or key(19) then dy=dy+1 end
 if btn(0) or key(23) then dy=dy-1 end
  
 if dx==0 then plr.sprite=1 end

 if dx~=0 or dy~=0 then
  local length=math.sqrt(dx*dx+dy*dy)
  
  plr.x=plr.x+(dx/length)*BASE_SPEED
  plr.y=plr.y+(dy/length)*BASE_SPEED
 end

 --screen edges
 if plr.x<48 then plr.x=48 end
 if plr.x>WIDTH-64 then plr.x=WIDTH-64 end
 if plr.y<0 then plr.y=0 end
 if plr.y>HEIGHT-16 then plr.y=HEIGHT-16 end
end

function draw_interface()
 rect(0,0,52,136,15)
 rect(188,0,52,136,15)

 print("HELLO WORLD!",84,84,6)
end

function plr_bullet()
 bul.y=bul.y-6
 bul.sprite=bul.sprite+0.5
 if bul.sprite>=10 then bul.sprite=7 end
 if btnp(4) or keyp(10) then bul.y=plr.y bul.x=plr.x+5 bul.sprite=7 sfx(00)
 end
end

function plr_flame()
 plr.flame=plr.flame+0.75
 if plr.flame>134 then plr.flame=128 end
end

function plr_muzzle()
 if btnp(4) or keyp(10) then plr.muzzle=8 end
 if plr.muzzle>0 then plr.muzzle=plr.muzzle-1 end
end

function plr_health()
 for i=1,3 do
  if plr.lives>=i then
   spr(98,i*18-18,11,0,2,0,0,1,1)
   else spr(99,i*18-18,11,0,2,0,0,1,1)
  end
 end
end

function update_starfield()
 if math.random()<0.08 then table.insert(starfield,{
 x=math.random(49,184),
 y=-24,
 spd=10,
 sprite=144
 })
end

for i=#starfield,1,-1 do
 local l=starfield[i]
 l.y=l.y+l.spd
 if l.y>HEIGHT then
  table.remove(starfield,i)
  end
 end
end

function draw_starfield()
 for _,l in ipairs(starfield) do
  spr(l.sprite,l.x,l.y,0,1,0,0,1,3)
 end
end
